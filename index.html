Enter another key." or "Auto-load failed. Verify your key to retry." end end local CoreGui = game:GetService('CoreGui'); local TweenService = game:GetService('TweenService'); local UserInputService = game:GetService('UserInputService'); local make("Frame", parent, props) end local function text(parent, value, props, class) props = props or {} props.Text = value props.BackgroundTransparency = props.BackgroundTransparency or 1 object, {CornerRadius = UDim.new(0, radius or 4)}) end local function stroke(object) return make("UIStroke", object, { Color = theme.Border, Thickness = 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, }) end local function animate(object, props) if not state.Alive then return end -- All calls for an object animate the same properties. if tweens[object] then tweens[object]:Cancel() end local tween = TweenService:Create(object, TweenInfo.new(0.14, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props) tweens[object] = tween tween:Play() end local gui = make("ScreenGui", CoreGui, { Name = "AzurixKeyGui", ResetOnSpawn = false, IgnoreGuiInset = false, DisplayOrder = 60, ZIndexBehavior = Enum.ZIndexBehavior.Sibling, }) local root = frame(gui, { Name = "Window", BackgroundColor3 = theme.Background, AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.new(1, -32, 0, 254), ClipsDescendants = true, }) make("UISizeConstraint", root, {MaxSize = Vector2.new(390, 254)}) round(root, 5) stroke(root).Color = theme.Accent local header = frame(root, { BackgroundTransparency = 1, Size = UDim2.new(1, -38, 0, Size = UDim2.new(1, -28, 0, 40), }) round(inputBox) local inputBorder = stroke(inputBox) local input = text(inputBox, "", { Position = UDim2.fromOffset(10, 0), Size = UDim2.new(1, -20, 1, 0), { Position = UDim2.fromOffset(14, 143), Size = UDim2.new(1, -28, 0, 30), TextColor3 = theme.Muted, TextWrapped = true, TextYAlignment = Enum.TextYAlignment.Top, }) local function statusColor() if statusKind == "error" then return Color3.fromRGB(225, 116, 116) end return statusKind == "accent" and theme.Accent or theme.Muted end local function setStatus(message, kind) if not state.Alive then return end statusKind = kind or "muted" status.Text = tostring(message or "") status.TextColor3 = statusColor() end local function paintButton(item) local amount = item.Down and 0.24 or item.Hover and 0.14 or item.Primary and 0.08 or 0 animate(item.Object, {BackgroundColor3 = theme.Row:Lerp(theme.Accent, amount)}) item.Object.TextColor3 = item.Primary and theme.Accent or theme.Text end local function button(caption, position, size, primary) local object = text(root, caption, { Position = position, Size = size, BackgroundTransparency = 0, BackgroundColor3 = theme.Row, TextXAlignment = local function cleanup() if not state.Alive then return end state.Alive = false requestId = requestId + 1 for _, connection in ipairs(connections) do connection:Disconnect() end for _, tween in function() destroy() end) local function submit() if state.Busy or not state.Alive then return end local key = input.Text:match("^%s*(.-)%s*$") if key == "" then setStatus("Enter your key first.", return end local function failed(message) if not state.Alive then return end state.Busy = false input.TextEditable = true verify.Text = "Verify" setStatus(message, "error") end if not fetched or type(source) ~= "string" or source == "" then failed("Download failed. Check SCRIPT_URL and try again.") return end local compiled, loader = pcall(loadstring, source, "AzurixLoader") if not compiled or type(loader) ~= "function" then failed("The script URL did not return valid Lua.") return end -- The loaded script must validate this key with your key service. local environment = getgenv() environment[KEY_VARIABLE] = key lp_key = key local ran, result = pcall(loader) if not state.Alive then return end if not ran then failed("Script failed to run. Check your key and try again.") else destroy() end end) end connect(verify.Activated, submit) connect(input.Focused, function() focused = true animate(inputBorder, {Color = theme.Accent}) end) connect(input.FocusLost, function(enter) focused = false animate(inputBorder, {Color = theme.Border}) if enter then submit() end end) connect(getKey.Activated, function() if state.Busy then return end if KEY_URL == "" then setStatus("Set KEY_URL at the top of the script.", "error") return end if type(setclipboard) ~= "function" then setStatus("Clipboard is unavailable in this environment.", "error") return end local copied = pcall(setclipboard, KEY_URL) setStatus(copied and "Key link copied. Paste into your browser." or "Unable to copy the key link.", copied and "accent" or "error") end) local dragInput, dragStart, startPosition connect(header.InputBegan, function(event) if event.UserInputType == Enum.UserInputType.MouseButton1 or event.UserInputType == Enum.UserInputType.Touch then dragInput, dragStart, startPosition = event, event.Position, root.Position end end) connect(UserInputService.InputChanged, function(event) if dragInput and (event == dragInput or (dragInput.UserInputType == Enum.UserInputType.MouseButton1 and event.UserInputType == Enum.UserInputType.MouseMovement)) then local delta = event.Position - dragStart root.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y) end end) connect(UserInputService.InputEnded, function(event) if event == dragInput then dragInput = nil end end) if autoLoadError then input.Text = suppliedKey setStatus(autoLoadError, "error") end
